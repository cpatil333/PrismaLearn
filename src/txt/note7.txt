///Query Extension
const prismas = new PrismaClient().$extends({
  model: {
    employee: {
      async findHighSalaryEmployees() {
        // Access the original Employee model methods
        const context = Prisma.getExtensionContext(this);

        // Fetch employees earning more than 50000
        const employees = await (context as any).findMany({
          where: {
            salary: {
              gt: 50000,
            },
          },
        });

        // Return the result
        return employees;
      },
    },
  },
});

const prisma = new PrismaClient();

const prismas = prisma.$extends({
  model: {
    employee: {
      async findEmployeesWithDepartment() {
        const context = Prisma.getExtensionContext(this);

        return (context as any).findMany({
          include: {
            department: {
              select: {
                name: true,
              },
            },
          },
        });

    },
  },

greate than and less than salary
const prismas = new PrismaClient().$extends({
  model: {
    employee: {
      async findEmployeesBySalary(minSalary: number, maxSalary: number) {
        // Access the original Employee model methods
        const context = Prisma.getExtensionContext(this);

        // Fetch employees earning more than 50000
        const employees = await (context as any).findMany({
          where: {
            AND: [
              { salary: { gte: minSalary } },
              { salary: { lte: maxSalary } },
            ],
          },
        });

        // Return the result
        return employees;
      },
    },
  },
});

    OR: [

  },
});
