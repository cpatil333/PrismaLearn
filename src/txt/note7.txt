const prismas = new PrismaClient().$extends({
  model: {
    employee: {
      async findHighSalaryEmployees() {
        // Access the original Employee model methods
        const context = Prisma.getExtensionContext(this);

        // Fetch employees earning more than 50000
        const employees = await (context as any).findMany({
          where: {
            salary: {
              gt: 50000,
            },
          },
        });

        // Return the result
        return employees;
      },
    },
  },
});

const prismas = new PrismaClient().$extends({
  model: {
    employee: {
      async findEmployeesBySalary(minSalary: number, maxSalary: number) {
        // Access the original Employee model methods
        const context = Prisma.getExtensionContext(this);

        // Fetch employees earning more than 50000
        const employees = await (context as any).findMany({
          where: {
            AND: [
              { salary: { gte: minSalary } },
              { salary: { lte: maxSalary } },
            ],
          },
        });

        // Return the result
        return employees;
      },
    },
  },
});

///// Decrement / increment

const prismas = new PrismaClient().$extends({
  query: {
    employee: {
      async update({ args, query }) {
        const employee = await prisma.employee.findUnique({
          where: {
            id: args.where.id,
          },
          select: {
            salary: true,
          },
        });

        let newSalary = employee?.salary;
        const salary = args.data.salary;

        if (typeof salary === "number") {
          newSalary = salary;
        } else if (
          salary &&
          "increment" in salary &&
          typeof salary.increment === "number"
        ) {
          newSalary = Number(employee?.salary) + salary.increment;
        } else if (
          salary &&
          "decrement" in salary &&
          typeof salary.decrement === "number"
        ) {
          newSalary = Number(employee?.salary) - salary.decrement;
        }
        // 2. Validate before updating
        if (newSalary !== undefined && newSalary < 20000) {
          throw new Error("Employee salary must be at least 20000");
        }
        console.log(employee);

        // We'll add validation here next.
        return query(args);
      },
    },
  },
});

  console.log(`Operation: ${operation}`);
  console.log(`Employee: ${identifier}`);
  console.log(`Timestamp: ${new Date().toISOString()}`);
}
