const prismas = new PrismaClient().$extends({
  result: {
    employee: {
      annualSalary: {
        needs: {
          salary: true,
        },
        compute(employee) {
          return employee.salary * 12;
        },
      },
    },
  },
});

async function main() {
  await prismas.$connect();
  console.log("Database connected successfully");

  const employee = await prismas.employee.findMany({
    select: {
      id: true,
      name: true,
      salary: true,
      annualSalary: true,
    },
  });

  console.log(employee);
}

const prismas = new PrismaClient().$extends({

    department: {
      departmentName: {
        needs: {
          name: true,
        },
        compute(department) {
          return department.name;
        },
      },
    },
  },
});

  await prismas.$connect();
  console.log("Database connected successfully");

  const employee = await prismas.employee.findMany({
    select: {
      id: true,

      salary: true,
      department: {
        select: {
          departmentName: true,
        },
      },
    },
  });

  const employeeData = employee.map(({ department, ...emp }) => ({
    ...emp,
    department: department.departmentName,
  }));

  console.log(employeeData);
}

const prismas = new PrismaClient().$extends({
  client: {
    async departmentStats(departmentId: number) {
      return prisma.employee.aggregate({
        where: { departmentId },
        _count: true,
        _avg: { salary: true },
      });
    },

    async highSalaryEmployees(departmentId: number) {
      return prisma.employee.count({
        where: {
          departmentId,
          salary: { gt: 50000 },
        },
      });
    },
  },
});

async function main() {
  await prismas.$connect();
  console.log("Database connected successfully");

  const [departmentStats, highSalaryEmployees] = await Promise.all([
    prismas.departmentStats(1),
    prismas.highSalaryEmployees(1),
  ]);

  console.log("Department Stats", departmentStats);
  console.log("High Salary Employees :", highSalaryEmployees);
}

      });
