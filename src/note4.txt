async function getAllDepartments() {
  const departments = await prisma.department.findMany({
    include: {
      employees: {
        select: {
          name: true,
          salary: true,
        },
      },
    },
  });
  return departments;
}

async function getDepartmentsWithEmployeesSalary() {
  const department = await prisma.department.findMany({
    select: {
      name: true,
      employees: {
        where: {
          salary: { gt: 50000 },
        },
        select: {
          name: true,
          salary: true,
        },
      },
    },
  });
  return department;
}

        where: {
          salary: { gt: minSalary },

        },
      },
    },
  });

}

async function getDepartmentWithEmployees(departmentId: number) {
  const depatment = await prisma.department.findUnique({
    where: {
      id: departmentId,
    },
    select: {
      name: true,
      employees: {
        select: {
          name: true,
          email: true,
          salary: true,
        },
      },
    },
  });

  if (depatment === null) {
    throw new Error("Department not found");
  }
  return depatment;
}
